<?php

namespace App\Http\Resources\DataEntry;

use App\Http\Resources\Global\Other\BasicResource;
use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;

class UserSpaceResource extends JsonResource
{
    /**
     * Transform the resource into an array.
     *
     * @return array<string, mixed>
     */
    public function toArray(Request $request): array
    {
        return [
            'sector' => $this->whenLoaded('sector', fn () => new BasicResource($this->sector), ['id' => $this->sector_id]),
            'department' => $this->whenLoaded('department', fn () => new BasicResource($this->department), ['id' => $this->department_id]),
            'location' => $this->whenLoaded('location', fn () => new BasicResource($this->location), ['id' => $this->location_id]),
        ];
    }
}
