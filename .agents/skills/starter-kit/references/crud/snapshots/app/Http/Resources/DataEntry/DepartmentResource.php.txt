<?php

namespace App\Http\Resources\DataEntry;

use App\Enum\DataEntry\DepartmentTypeEnum;
use App\Http\Resources\Global\Other\BasicUserResource;
use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;

class DepartmentResource extends JsonResource
{
    public function toArray(Request $request): array
    {
        return [
            'id' => $this->id,
            'is_active' => $this->is_active,
            'translation_name' => $this->name,
            'name' => $this->getTranslations('name'),
            'key' => $this->key,
            'type' => $this->type,
            'display_type' => DepartmentTypeEnum::resolve($this->type),
            'translation_description' => $this->description,
            'description' => $this->getTranslations('description'),

            'location_assignments' => $this->whenLoaded('locationAssignments', fn () => DepartmentLocationResource::collection($this->locationAssignments), []),
            'creator' => $this->whenLoaded('creator', fn () => new BasicUserResource($this->creator), ['id' => $this->created_by]),
            'created_at' => $this->created_at,
        ];
    }
}
